🐍 原始 Python 代码

def factor_volume_weighted_impact(df):
    df_copy = df.copy()
    ret = df_copy['close_price'].pct_change().fillna(0.0)
    abs_ret = ret.abs()
    volume = df_copy['volume']
    total_volume = volume.sum()
    if total_volume > 0:
        impact = (abs_ret * volume).sum() / total_volume
    else:
        impact = 0.0
    day = pd.Timestamp(df_copy['TradingDay'].iloc[0])
    out = pd.Series([impact], index=[day], name='factor_volume_weighted_impact')
    return out
