def factor_volume_weighted_squared_impact(df):
    df_copy = df.copy()
    ret = df_copy['close_price'].pct_change().fillna(0.0)
    sq_ret = ret ** 2
    volume = df_copy['volume']
    total_volume = volume.sum()
    if total_volume > 0:
        impact = (sq_ret * volume).sum() / total_volume
    else:
        impact = 0.0
    day = pd.Timestamp(df_copy['TradingDay'].iloc[0])
    out = pd.Series([impact], index=[day], name='factor_volume_weighted_squared_impact')
    return out
