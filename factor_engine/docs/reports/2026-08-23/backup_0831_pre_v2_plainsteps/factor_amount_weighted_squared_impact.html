🐍 原始 Python 代码

def factor_amount_weighted_squared_impact(df):
    df_copy = df.copy()
    ret = df_copy['close_price'].pct_change().fillna(0.0)
    sq_ret = ret ** 2
    amount = df_copy['amount']
    total_amount = amount.sum()
    if total_amount > 0:
        impact = (sq_ret * amount).sum() / total_amount
    else:
        impact = 0.0
    day = pd.Timestamp(df_copy['TradingDay'].iloc[0])
    out = pd.Series([impact], index=[day], name='factor_amount_weighted_squared_impact')
    return out
