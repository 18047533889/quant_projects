"kw">class="kw">def factor_path_efficiency_tail_mean_smooth(df): """Reward persistent directional price paths ">class="kw">while using a smoother downside-fragility modifier.""" df_copy = df.copy() close = df_copy["close"].replace(0.0, np.nan) net_move = close - close.shift(126) path_length = close.diff().abs().rolling(126, min_periods=126).sum() efficiency = net_move / path_length.replace(0.0, np.nan) recent_move = close - close.shift(63) recent_length = close.diff().abs().rolling(63, min_periods=63).sum() recent_efficiency = recent_move / recent_length.replace(0.0, np.nan) ret = close / close.shift(1) - 1.0 downside_pressure = (-ret.clip(upper=0.0)).rolling(20, min_periods=20).mean() upside_pressure = ret.clip(lower=0.0).rolling(20, min_periods=20).mean() baseline = ret.abs().rolling(20, min_periods=20).mean().clip(lower=1e-6) downside_fragility = ((downside_pressure - upside_pressure) / baseline).clip(lower=0.0, upper=2.0) signal = efficiency * recent_efficiency.abs() * (1.0 - 0.25 * downside_fragility) df_copy["factor_path_efficiency_tail_mean_smooth"] = signal.replace([np.inf, -np.inf], np.nan) "kw">class="kw">return df_copy["factor_path_efficiency_tail_mean_smooth"]
