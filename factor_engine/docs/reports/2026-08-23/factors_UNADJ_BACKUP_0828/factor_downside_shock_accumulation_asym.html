def factor_downside_shock_accumulation_asym(df):
    """Idea: Repeated prior downside shocks accumulate fragility, and the current downside-volatility premium is amplified by that accumulated shock count.

    Logic: A downside shock is a close-to-close return below -2%. We count such shocks over the prior 60 trading days (excluding today). The current downside volatility premium is measured as the difference between downside and upside volatility from a 20-day return decomposition. Multiplying the premium by the prior shock count makes the volatility-asymmetry response path-dependent: the same downside-volatility level carries more fragility information after repeated adverse sessions.

    Sub-mechanism tags: fragility_accumulation, downside_vol_premium, shock_count, volatility_asymmetry

    Formula: ret = pct_change(close); shock = 1[ret < -0.02]; prior = rolling_sum(shift(shock), 60); premium = max(down_vol - up_vol, 0); factor = prior * premium
    """
    df_copy = df.copy()
    ret = df_copy["close"].pct_change()
    upside_vol, downside_vol = alpha_tools.decompose_return_volatility_direction(ret, min_periods=10, window=20)
    downside_premium = (downside_vol - upside_vol).clip(lower=0.0)
    down_shock = (ret < -0.02).astype(float)
    prior_shocks = down_shock.shift(1).rolling(60, min_periods=20).sum()
    factor = prior_shocks * downside_premium
    factor = factor.replace([np.inf, -np.inf], np.nan)
    df_copy.loc[:, "factor_downside_shock_accumulation_asym"] = factor
    return df_copy["factor_downside_shock_accumulation_asym"]
