"kw">class="kw">def factor_asym_vol_weighted(df): df_copy = df.copy() ret = df_copy["close"].pct_change() down = ret < 0 up = ret >= 0 range_ = df_copy["high"] - df_copy["low"] vol = df_copy["volume"] down_range_vol = (range_ * vol).where(down, 0) down_vol_sum = vol.where(down, 0).rolling(20, min_periods=10).sum() down_range_vol_sum = down_range_vol.rolling(20, min_periods=10).sum() down_avg = down_range_vol_sum / down_vol_sum.replace(0, np.nan) up_range_vol = (range_ * vol).where(up, 0) up_vol_sum = vol.where(up, 0).rolling(20, min_periods=10).sum() up_range_vol_sum = up_range_vol.rolling(20, min_periods=10).sum() up_avg = up_range_vol_sum / up_vol_sum.replace(0, np.nan) ratio = down_avg / up_avg.replace(0, np.nan) factor = np.log(ratio.clip(lower=1e-6)) df_copy["factor_asym_vol_weighted"] = factor "kw">class="kw">return df_copy["factor_asym_vol_weighted"]
