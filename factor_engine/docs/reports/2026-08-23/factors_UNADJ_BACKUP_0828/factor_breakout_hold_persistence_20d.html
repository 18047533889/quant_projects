def factor_breakout_hold_persistence_20d(df):
    '''
    Idea: A genuine level repricing is confirmed when, after breaking the prior 20-day high,
    the stock persistently closes in the upper quartile of the post-breakout channel instead of
    fading back toward the broken level.

    Logic: The prior 20-day high acts as the new floor. The 10-day persistence of closes in the
    upper quartile of the post-breakout channel captures the buying-pressure mechanism from the
    persistent-upper-quartile parent, while the ATR-normalized distance above the broken level,
    directional efficiency, and volume momentum retain the breakout-decisiveness mechanism from
    the efficiency parent. Persistent holds above the level receive high positive values; failed
    attempts that close back below resistance collapse to zero.

    Sub-mechanism tags: post-breakout-hold, upper-quartile-persistence, directional-efficiency, volume-confirmation

    Formula: hold_persistence * (close-prior_high)/ATR * (0.5+0.5*efficiency) * volume_momentum
    '''
    df_copy = df.copy()
    atr = alpha_tools.average_true_range(df_copy, window=14).replace(0.0, np.nan)
    resistance = df_copy['high'].rolling(20).max().shift(1)
    channel_high = df_copy['high'].rolling(10, min_periods=5).max()
    channel_range = (channel_high - resistance).replace(0.0, np.nan)
    hold_loc = (df_copy['close'] - resistance) / channel_range
    hold_flag = ((hold_loc >= 0.75) & (df_copy['close'] > resistance)).astype(float)
    hold_persistence = hold_flag.rolling(10, min_periods=5).sum() / 10.0
    position = ((df_copy['close'] - resistance) / atr).clip(0.0, 5.0)
    efficiency = alpha_tools.directional_efficiency(df_copy, window=10)
    vol_conf = alpha_tools.volume_momentum_ratio(df_copy['volume'], long_window=20, short_window=5).clip(0.5, 2.0)
    factor = position * hold_persistence * (0.5 + 0.5 * efficiency.clip(-1.0, 1.0)) * vol_conf
    factor = factor.replace([np.inf, -np.inf], np.nan)
    df_copy['factor_breakout_hold_persistence_20d'] = factor
    return df_copy['factor_breakout_hold_persistence_20d']
