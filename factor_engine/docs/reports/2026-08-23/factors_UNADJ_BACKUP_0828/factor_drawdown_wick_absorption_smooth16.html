"kw">class="kw">def factor_drawdown_wick_absorption_smooth16(df): """Measure smoothed lower-wick absorption ">class="kw">while price remains underwater.""" df_copy = df.copy() high = df_copy["high"] low = df_copy["low"] open_ = df_copy["open"] close = df_copy["close"] eps = np.finfo(float).eps safe_close = close.clip(lower=eps) bar_range = (high - low).replace(0.0, np.nan) wick_base = np.minimum(open_, close) lower_wick = ((wick_base - low) / bar_range).clip(lower=0.0, upper=1.0) close_location = ((close - low) / bar_range).clip(lower=0.0, upper=1.0) absorption = (lower_wick * close_location).rolling(16, min_periods=16).mean() high_water = safe_close.rolling(50, min_periods=50).max().clip(lower=eps) depth = (1.0 - safe_close / high_water).clip(lower=0.0) prior_close = safe_close.shift(1) downside_damage = ((-np.log(safe_close / prior_close)).clip(lower=0.0)).rolling(12, min_periods=12).sum() signal = depth * absorption / (1.0 + downside_damage) df_copy["factor_drawdown_wick_absorption_smooth16"] = signal.replace([np.inf, -np.inf], np.nan) "kw">class="kw">return df_copy["factor_drawdown_wick_absorption_smooth16"]
