"kw">class="kw">def factor_drawdown_wick_absorption_tail_guard(df): """Measure underwater lower-wick absorption with bounded drawdown-conditioned tail-risk attenuation.""" df_copy = df.copy() high = df_copy["high"] low = df_copy["low"] open_ = df_copy["open"] close = df_copy["close"].replace([np.inf, -np.inf], np.nan) eps = np.finfo(float).eps safe_close = close.clip(lower=eps) bar_range = (high - low).replace(0.0, np.nan) wick_base = np.minimum(open_, close) lower_wick = ((wick_base - low) / bar_range).clip(lower=0.0, upper=1.0) close_location = ((close - low) / bar_range).clip(lower=0.0, upper=1.0) absorption = (lower_wick * close_location).rolling(16, min_periods=16).mean() high_water = safe_close.rolling(50, min_periods=50).max().clip(lower=eps) depth = (1.0 - safe_close / high_water).clip(lower=0.0) prior_close = safe_close.shift(1) downside_damage = (-np.log(safe_close / prior_close)).clip(lower=0.0).rolling(12, min_periods=12).sum() ret = safe_close.pct_change().replace([np.inf, -np.inf], np.nan) downside_energy = ret.clip(upper=0.0).pow(2) upside_energy = ret.clip(lower=0.0).pow(2) rolling_peak = safe_close.rolling(20, min_periods=20).max().clip(lower=eps) local_drawdown = (1.0 - safe_close / rolling_peak).clip(lower=0.0) weighted_downside = (downside_energy * (1.0 + local_drawdown)).rolling(20, min_periods=20).mean() upside_baseline = upside_energy.rolling(20, min_periods=20).mean() total_energy = (downside_energy + upside_energy).rolling(20, min_periods=20).mean().replace(0.0, np.nan) tail_fragility = ((weighted_downside - upside_baseline) / total_energy).clip(lower=0.0, upper=1.0) signal = depth * absorption / (1.0 + downside_damage) signal = signal * (1.0 - 0.25 * tail_fragility) df_copy["factor_drawdown_wick_absorption_tail_guard"] = signal.replace([np.inf, -np.inf], np.nan) "kw">class="kw">return df_copy["factor_drawdown_wick_absorption_tail_guard"]
