"kw">class="kw">def factor_impact_weighted_asymmetry_normalized(df): """Normalize impact-weighted downside asymmetry by total impact-weighted volatility.""" df_copy = df.copy() close = df_copy["close"] volume = df_copy["volume"] safe_close = close.where(close > 0.0) prev_close = safe_close.shift(1) safe_volume = volume.where(volume > 0.0) ret = (safe_close / prev_close - 1.0).replace([np.inf, -np.inf], np.nan) dollar_volume = (prev_close * safe_volume).replace([np.inf, -np.inf], np.nan) impact = (ret.abs() / dollar_volume).replace([np.inf, -np.inf], np.nan) impact_base = impact.rolling(20, min_periods=10).median() impact_base = impact_base.where(np.isfinite(impact_base) & (impact_base > 0.0)) relative_impact = (impact / impact_base).replace([np.inf, -np.inf], np.nan).clip(upper=10.0) downside = ret.clip(upper=0.0).pow(2) * relative_impact upside = ret.clip(lower=0.0).pow(2) * relative_impact asymmetry = downside.rolling(20, min_periods=10).mean() - upside.rolling(20, min_periods=10).mean() total_stress = (downside + upside).rolling(20, min_periods=10).mean() df_copy["factor_impact_weighted_asymmetry_normalized"] = (asymmetry / total_stress.replace(0.0, np.nan)).replace([np.inf, -np.inf], np.nan) "kw">class="kw">return df_copy["factor_impact_weighted_asymmetry_normalized"]
