"kw">class="kw">def factor_directional_imbalance_trend_aligned(df): df_copy = df.copy() prev_close = df_copy["str">'close'].shift(1) buy = (df_copy["str">'high'] - prev_close) * df_copy["str">'volume'] sell = (prev_close - df_copy["str">'low']) * df_copy["str">'volume'] window = 20 net = (buy.rolling(window).sum() - sell.rolling(window).sum()) / df_copy["str">'volume'].rolling(window).sum().replace(0, np.nan) adx = pd.Series(talib.ADX(df_copy["str">'high'], df_copy["str">'low'], df_copy["str">'close'], timeperiod=50), index=df_copy.index) pdi = pd.Series(talib.PLUS_DI(df_copy["str">'high'], df_copy["str">'low'], df_copy["str">'close'], timeperiod=50), index=df_copy.index) mdi = pd.Series(talib.MINUS_DI(df_copy["str">'high'], df_copy["str">'low'], df_copy["str">'close'], timeperiod=50), index=df_copy.index) direction = np.sign(pdi - mdi) factor = np.where(adx > 25, net * direction, net * 0.5) factor = pd.Series(factor, index=df_copy.index).ewm(span=10).mean() factor.name = "factor_directional_imbalance_trend_aligned" "kw">class="kw">return factor
