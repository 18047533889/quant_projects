🐍 原始 Python 代码 (factor_delivery)

"kw">class="kw">def factor_improving_quality_value(df): df_copy = df.copy() pb = df_copy["pb_lf"] cheapness = 1.0 / pb.replace(0, np.nan) cheapness = cheapness.where(pb > 0, np.nan) roe = df_copy["roe_ttm2"] / 100.0 roe_delta = roe - roe.shift(4) rev_growth = df_copy["qfa_yoygr"] / 100.0 roe_delta_clipped = roe_delta.clip(-0.9, 10) rev_growth_clipped = rev_growth.clip(-0.9, 10) composite = cheapness * (1 + roe_delta_clipped) * (1 + rev_growth_clipped) composite = composite.replace([np.inf, -np.inf], np.nan) composite.name = "factor_improving_quality_value" "kw">class="kw">return composite
