def factor_failed_retest_reversion_20_5(df):
    """Idea: After an upside breakout, a retest that fails to hold the prior 20-day high (price touches the boundary and closes back below it) carries reversion information: repeated failed retests with elevated volume and closes near the lows separate a failed attempt that reverts into the range from a genuine level-hold.

    Logic: recent_breakout flags any close above the prior 20-day high in the last five days (shifted). retest_failure marks days where low <= prior_high and close < prior_high. failure_depth is the distance of the close below the prior high normalized by ATR20. rejection is 1 - close location, so closes near the low receive the strongest rejection weight. volume ratio is capped at 3. The daily failure score is summed over five days to capture repeated failed attempts. The raw signal is negative, expressing expected reversion, and is only active after a recent upside breakout.

    Sub-mechanism tags: failed_retest, breakout_reversion, close_rejection, volume_confirmation
    """
    df_copy = df.copy()
    prior_high = df_copy["high"].rolling(20, min_periods=20).max().shift(1)
    recent_breakout = (df_copy["close"] > prior_high).astype(float).rolling(5, min_periods=1).max().shift(1)
    atr = alpha_tools.average_true_range(df_copy, window=20)
    day_range = (df_copy["high"] - df_copy["low"]).replace(0.0, np.nan)
    close_loc = ((df_copy["close"] - df_copy["low"]) / day_range).clip(0, 1)
    rejection = (1.0 - close_loc).fillna(0.0)
    failure_depth = ((prior_high - df_copy["close"]).clip(lower=0) / atr.replace(0.0, np.nan)).fillna(0.0)
    retest_failure = ((df_copy["low"] <= prior_high) & (df_copy["close"] < prior_high)).astype(float)
    vol_ratio = (df_copy["volume"] / df_copy["volume"].rolling(20, min_periods=20).mean().replace(0.0, np.nan)).clip(upper=3.0).fillna(1.0)
    daily_failure = retest_failure * failure_depth * rejection * vol_ratio
    recent_failure = daily_failure.rolling(5, min_periods=1).sum().fillna(0.0)
    signal = -recent_breakout * recent_failure
    signal = signal.where(np.isfinite(signal), 0.0)
    signal.name = "factor_failed_retest_reversion_20_5"
    return signal
